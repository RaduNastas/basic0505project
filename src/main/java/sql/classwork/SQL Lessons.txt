Update lista studenti orasele in care locuesc.
UPDATE students
SET city = CASE 
    WHEN first_name = 'Anna'    AND last_name = 'Petrova'    THEN 'Berlin'
    WHEN first_name = 'Ivan'    AND last_name = 'Sidorov'    THEN 'Munich'
    WHEN first_name = 'Maria'   AND last_name = 'Ivanova'    THEN 'Hamburg'
    WHEN first_name = 'Dmitry'  AND last_name = 'Smirnov'    THEN 'Berlin'
    WHEN first_name = 'Olga'    AND last_name = 'Kuznetsova' THEN 'Cologne'
    WHEN first_name = 'Nikolay' AND last_name = 'Popov'      THEN 'Frankfurt'
    WHEN first_name = 'Elena'   AND last_name = 'Volkova'    THEN 'Berlin'
    WHEN first_name = 'Sergey'  AND last_name = 'Morozov'    THEN 'Stuttgart'
    WHEN first_name = 'Irina'   AND last_name = 'Sokolova'   THEN 'Dresden'
    WHEN first_name = 'Pavel'   AND last_name = 'Lebedev'    THEN 'Leipzig'
END;
